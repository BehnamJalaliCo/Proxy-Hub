   import asyncio
   from urllib.parse import unquote
   from proxyhub.db.session import get_engine
   from proxyhub.reality.links import provide_links
   async def main():
       engine = get_engine()
       async with engine.connect() as db:
           names = sorted({(l.slot, unquote(l.url.rsplit("#", 1)[1]).split(" ")[1]) for l in await provide_links(db) if l.kind == "reality" and l.status == "active"})

   asyncio.run(main())
